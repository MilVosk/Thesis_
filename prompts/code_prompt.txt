    pass

    """Measured parameters or traits (volume, morphology, biomass)."""

class LOCATION(Entity):
    """Geographic places without coordinates (China, Amazon basin)."""
    pass


class MATTER(Entity):
    """Chemical/biological compounds or natural elements (carbon, sediment)."""
    pass


class Relation:
    def __init__(self, head_entity: Entity, tail_entity: Entity):
        self.head_entity = head_entity
        self.tail_entity = tail_entity

    """Entity A has/possesses Quality B or is composed of B."""
    # Allowed pairs: ORGANISM-ENVIRONMENT, ORGANISM-QUALITY, ENVIRONMENT-QUALITY,
    # MATTER-QUALITY, ENVIRONMENT-ORGANISM, QUALITY-ORGANISM, QUALITY-ENVIRONMENT, QUALITY-MATTER
    pass


class OccurIn(Relation):
    """Entity A occurs/exists within Entity B."""
    # Allowed pairs: MATTER-ENVIRONMENT, ENVIRONMENT-MATTER, ORGANISM-LOCATION,
    # LOCATION-ORGANISM, ORGANISM-ORGANISM, PHENOMENA-LOCATION, LOCATION-PHENOMENA,
    # ENVIRONMENT-LOCATION, LOCATION-ENVIRONMENT
    pass


class Influence(Relation):
    """Entity A influences/affects Entity B."""
    # Allowed pairs: ORGANISM-PHENOMENA, PHENOMENA-ORGANISM, ORGANISM-MATTER,
    # MATTER-ORGANISM, PHENOMENA-PHENOMENA, PHENOMENA-QUALITY, QUALITY-PHENOMENA,
    # PHENOMENA-ENVIRONMENT, ENVIRONMENT-PHENOMENA, QUALITY-QUALITY
    pass
